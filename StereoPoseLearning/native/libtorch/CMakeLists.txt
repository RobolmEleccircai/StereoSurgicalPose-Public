cmake_minimum_required(VERSION 3.18)
project(stereo_libtorch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(STEREO_PROFILE_RANGES "Enable profiler stage labels in a separate diagnostic build" OFF)
option(STEREO_WITH_TENSORRT "Build a separate native TensorRT visual frontend adapter" OFF)
option(STEREO_OPTIMIZED "Build separate optimized CUDA runtime" OFF)
option(STEREO_FUSIONS "Build independently selectable fused CUDA operators" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c
  "import pathlib,torch; print(pathlib.Path(torch.__file__).parent); print(int(torch.compiled_with_cxx11_abi()))"
  OUTPUT_VARIABLE TORCH_INFO OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\n" ";" TORCH_INFO "${TORCH_INFO}")
list(GET TORCH_INFO 0 TORCH_ROOT)
list(GET TORCH_INFO 1 TORCH_ABI)
Python3_add_library(stereo_libtorch MODULE model.cpp bindings.cpp preprocess.cpp)
target_include_directories(stereo_libtorch PRIVATE "${TORCH_ROOT}/include" "${TORCH_ROOT}/include/torch/csrc/api/include")
target_compile_definitions(stereo_libtorch PRIVATE _GLIBCXX_USE_CXX11_ABI=${TORCH_ABI})
if(STEREO_PROFILE_RANGES)
  target_compile_definitions(stereo_libtorch PRIVATE STEREO_PROFILE_RANGES=1)
endif()
target_link_libraries(stereo_libtorch PRIVATE "${TORCH_ROOT}/lib/libtorch.so" "${TORCH_ROOT}/lib/libtorch_cpu.so"
  "${TORCH_ROOT}/lib/libtorch_cuda.so" "${TORCH_ROOT}/lib/libtorch_python.so" "${TORCH_ROOT}/lib/libc10.so")
set_target_properties(stereo_libtorch PROPERTIES BUILD_RPATH "${TORCH_ROOT}/lib" CXX_VISIBILITY_PRESET hidden)
if(STEREO_WITH_TENSORRT)
  find_path(TENSORRT_INCLUDE NvInfer.h REQUIRED)
  find_library(TENSORRT_LIBRARY NAMES nvinfer libnvinfer.so.10 REQUIRED)
  find_path(CUDA_RUNTIME_INCLUDE cuda_runtime_api.h REQUIRED)
  target_sources(stereo_libtorch PRIVATE trt_visual.cpp)
  target_include_directories(stereo_libtorch PRIVATE ${TENSORRT_INCLUDE} ${CUDA_RUNTIME_INCLUDE})
  target_compile_definitions(stereo_libtorch PRIVATE STEREO_WITH_TENSORRT=1)
  target_link_libraries(stereo_libtorch PRIVATE ${TENSORRT_LIBRARY} "${TORCH_ROOT}/lib/libc10_cuda.so")
  if(NOT STEREO_OPTIMIZED)
    target_compile_definitions(stereo_libtorch PRIVATE STEREO_MODULE_NAME=stereo_libtorch_trt)
    set_target_properties(stereo_libtorch PROPERTIES OUTPUT_NAME stereo_libtorch_trt)
  endif()
endif()

if(STEREO_OPTIMIZED)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(stereo_libtorch PRIVATE position_cuda.cu)
  target_compile_definitions(stereo_libtorch PRIVATE STEREO_OPTIMIZED=1)
  target_link_libraries(stereo_libtorch PRIVATE CUDA::cudart "${TORCH_ROOT}/lib/libc10_cuda.so")
  target_compile_options(stereo_libtorch PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  if(STEREO_FUSIONS)
    target_compile_definitions(stereo_libtorch PRIVATE STEREO_MODULE_NAME=stereo_libtorch_fused)
  else()
    target_compile_definitions(stereo_libtorch PRIVATE STEREO_MODULE_NAME=stereo_libtorch_opt)
  endif()
  set_target_properties(stereo_libtorch PROPERTIES OUTPUT_NAME stereo_libtorch_opt CUDA_STANDARD 17)
endif()

if(STEREO_FUSIONS)
  if(NOT STEREO_OPTIMIZED OR NOT STEREO_WITH_TENSORRT)
    message(FATAL_ERROR "STEREO_FUSIONS requires optimized TensorRT build")
  endif()
  target_sources(stereo_libtorch PRIVATE preprocess_cuda.cu)
  target_compile_definitions(stereo_libtorch PRIVATE STEREO_FUSIONS=1)
  set_target_properties(stereo_libtorch PROPERTIES OUTPUT_NAME stereo_libtorch_fused)
endif()
