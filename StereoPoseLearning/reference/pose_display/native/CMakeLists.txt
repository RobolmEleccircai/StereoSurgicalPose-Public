cmake_minimum_required(VERSION 3.16)
project(surgical_overlay LANGUAGES CXX)
option(STEREO_PROFILE_RANGES "Enable CPU stage timings in a diagnostic build" OFF)
option(STEREO_OPTIMIZED_OVERLAY "Cache panels and paint eyes on persistent workers" OFF)
find_package(Threads REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc freetype imgcodecs)
add_library(surgical_overlay SHARED overlay.cpp)
if(STEREO_PROFILE_RANGES)
  target_compile_definitions(surgical_overlay PRIVATE STEREO_PROFILE_RANGES=1)
endif()
if(STEREO_OPTIMIZED_OVERLAY)
  target_compile_definitions(surgical_overlay PRIVATE STEREO_OPTIMIZED_OVERLAY=1)
endif()
target_link_libraries(surgical_overlay PRIVATE Threads::Threads)
target_compile_features(surgical_overlay PRIVATE cxx_std_17)
target_compile_options(surgical_overlay PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(surgical_overlay PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(surgical_overlay PRIVATE ${OpenCV_LIBS})
set_target_properties(surgical_overlay PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_executable(overlay_pair render_pair.cpp)
target_compile_features(overlay_pair PRIVATE cxx_std_17)
target_include_directories(overlay_pair PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(overlay_pair PRIVATE surgical_overlay ${OpenCV_LIBS})
